int n13(int n){
    int k = 0;
    for (int i=2; i<n; i++){
        if (n % i == 0) k++;
    }
    return k;
}

string n21(int n){
    int k = 0;
    while (n/10 > 0){
        int a = n%10;
        int b = (n/10)%10;
        if (a <= b){
            k = 1;
            break;
        }
        n = n / 10;
    }
    if (k) return "не возрастает ";
    else return "возрастает ";
}

string n22(string n){
    int d = 0;
    int o = 0;
    int m = 0;
    for(char s : n){
        if (s == '!') break;
        if (s == 'd' || s == 'D') d++;
        else if (s == 'o' || s == 'O') o++;
        else if (s == 'm' || s == 'M') m++;
    }
    if (d > 0 && o > 0 && m > 0) return "можно";
    else return "нельзя";
}

bool check(int n) {
    if (n <= 1) return false;
    if (n == 2) return true;
    if (n % 2 == 0) return false;
    for (int i = 3; i * i <= n; i += 2) {
        if (n % i == 0) return false;
    }
    return true;
}
