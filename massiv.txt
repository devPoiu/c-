void replace(int a[], int n, int c=0 ){

int summa(int a[], int n, int c=1 ){
    int sum=0;
    for(int i=0;i<n;i++)
        if(a[i] % c==0)
        sum+=a[i];
    return sum ;
}

void min(int arr[], int n){
    int min;
    for (int i = 0; i< n; i++){
        if (arr[i] >= 0){
            min = arr[i];
            break;
        }
    }
    for (int i = 0; i< n; i++){
        if ((arr[i]>= 0) && (arr[i] < min)){
            min = arr[i];
        }
    }
    for (int i = 0; i< n; i++){
        if (arr[i] <= 0){
            arr[i] = min;
        }
    }
}
